async function handleAddPhrase() { const input = document.getElementById('phraseInput'); const phrase = input.value.trim(); if (!phrase) return; unlockAudioOnIOS(); const btn = document.getElementById('addBtn'); btn.innerText = 'Translating...'; btn.disabled = true; try { const response = await fetch(SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain' }, body: JSON.stringify({ phrase: phrase, mode: currentMode }) }); const result = await response.json(); if (result.status === 'success') { const newCard = { english: result.english, french: result.french, slang: result.slang }; localPhrases.unshift(newCard); saveToLocalStorage(localPhrases); renderCards(localPhrases); input.value = ''; // Sequence playback: Standard -> Pause 1s -> Slang if (result.french && result.slang) { speakFrenchSequence(result.french, result.slang); } else if (result.french) { speakFrench(result.french, false); } else if (result.slang) { speakFrench(result.slang, false); } } else { alert('Translation error: ' + (result.message || 'Unknown error')); } } catch (error) { alert('Failed to connect to translation server.'); } finally { btn.innerText = 'Translate'; btn.disabled = false; } } // Speaks standard French, waits for audio completion, pauses 1s, then speaks slang function speakFrenchSequence(firstText, secondText) { // Native iOS App Handler if (window.webkit && window.webkit.messageHandlers && window.webkit.messageHandlers.nativeSpeak) { window.webkit.messageHandlers.nativeSpeak.postMessage("NORMAL|" + firstText); setTimeout(() => { window.webkit.messageHandlers.nativeSpeak.postMessage("NORMAL|" + secondText); }, 2500); // Fixed buffer delay for iOS Native bridge return; } // Web Speech API Handler if (!('speechSynthesis' in window)) return; if (!frenchVoice) loadVoices(); window.speechSynthesis.cancel(); let utterance1 = new SpeechSynthesisUtterance(firstText); utterance1.lang = 'fr-FR'; utterance1.rate = isIOS ? 0.52 : 0.88; if (frenchVoice) utterance1.voice = frenchVoice; utterance1.onend = function() { // 1000ms delay STARTS strictly after phrase 1 finishes setTimeout(() => { let utterance2 = new SpeechSynthesisUtterance(secondText); utterance2.lang = 'fr-FR'; utterance2.rate = isIOS ? 0.52 : 0.88; if (frenchVoice) utterance2.voice = frenchVoice; window.speechSynthesis.speak(utterance2); }, 1000); }; window.speechSynthesis.speak(utterance1); } function speakFrench(text, isSlow = false) { if (!text) return; if (window.webkit && window.webkit.messageHandlers && window.webkit.messageHandlers.nativeSpeak) { const payload = (isSlow ? "SLOW|" : "NORMAL|") + text; window.webkit.messageHandlers.nativeSpeak.postMessage(payload); return; } if (!('speechSynthesis' in window)) return; if (!frenchVoice) loadVoices(); window.speechSynthesis.cancel(); let utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'fr-FR'; if (isIOS) { utterance.rate = isSlow ? 0.25 : 0.52; } else { utterance.rate = isSlow ? 0.5 : 0.88; } utterance.pitch = 0.95; if (frenchVoice) { utterance.voice = frenchVoice; } window.speechSynthesis.speak(utterance); }
